## make a vector of terms to concatenate with AND

func_AND_vector <- function(v){
  pat <- paste0("(?=.*(?:", v, "))", collapse="")
  return(pat)
}


## make a vector of terms to concatenate with OR
func_OR_vector <- function(v){
  pat <- paste0(v, collapse = "|")
  return(pat)
}
