names(df)

cmt_mean_goal <- df %>%
  dplyr::select(Company, iso3,
                goal, Target_id, n_statement) %>%
  dplyr::filter(!goal %in% c('SDG', 'noSDG')) %>%
  arrange(goal, Company, iso3) %>%
  group_by(iso3, Company, goal) %>%
  ### the total number of commitment statement to each goal by each company
  dplyr::summarise(n_cmt = sum(n_statement, na.rm = T)) %>%
  ### the average number of commitment statement to each goal in each country
  ungroup() %>%
  group_by(iso3, goal) %>%
  dplyr::summarise(n_cmt_mean = mean(n_cmt, na.rm = T)) %>%
  arrange(iso3, goal)
